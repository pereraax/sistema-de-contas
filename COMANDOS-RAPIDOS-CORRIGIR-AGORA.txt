===========================================
COMANDOS RÁPIDOS - CORRIGIR ASSETS AGORA
===========================================

Execute estes comandos NO SSH, um por vez:

===========================================
1. ENCONTRAR CAMINHO DO PROJETO
===========================================

find /home -name "next.config.js" 2>/dev/null

# Anote o caminho que aparecer (ex: /home/u596588143/domains/plenipay.com/)

===========================================
2. VERIFICAR SE ARQUIVOS ESTÁTICOS EXISTEM
===========================================

# Substitua pelo caminho encontrado acima
cd /home/u596588143/domains/plenipay.com
ls -la .next/static

# Se não existir, fazer build:
npm run build
pm2 restart sistema-contas

===========================================
3. EDITAR CONFIGURAÇÃO NGINX
===========================================

nano /etc/nginx/sites-available/plenipay

# Cole a configuração abaixo (ajuste o caminho se necessário)
# Salvar: Ctrl+X, Y, Enter

===========================================
CONFIGURAÇÃO NGINX (COLE ISSO):
===========================================

server {
    listen 80;
    server_name plenipay.com www.plenipay.com 31.97.27.20;

    location /_next/static {
        alias /home/u596588143/domains/plenipay.com/.next/static;
        expires 365d;
        add_header Cache-Control "public, immutable";
    }

    location ~* \.(jpg|jpeg|png|gif|ico|svg|woff|woff2|ttf|eot|css|js)$ {
        root /home/u596588143/domains/plenipay.com/public;
        expires 30d;
        add_header Cache-Control "public";
    }

    location / {
        proxy_pass http://localhost:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_cache_bypass $http_upgrade;
    }
}

===========================================
4. TESTAR E REINICIAR NGINX
===========================================

nginx -t
systemctl restart nginx
